cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(VirtualDisplayStudio VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with Visual Studio 2022 C++ tools on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(DISPLAY_ASAN "Instrument protocol tests with AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
add_executable(VirtualDisplayStudio WIN32 App/Main.cpp App/Display.cpp)
target_link_libraries(VirtualDisplayStudio PRIVATE setupapi user32 gdi32 shell32 advapi32 gdiplus)
target_link_options(VirtualDisplayStudio PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(DisplayControl App/Console.cpp App/Display.cpp)
target_link_libraries(DisplayControl PRIVATE setupapi)
add_executable(DisplayProtocolTests Tests/ProtocolTests.cpp)
target_include_directories(DisplayProtocolTests PRIVATE App)
if(DISPLAY_ASAN)
 target_compile_options(DisplayProtocolTests PRIVATE /fsanitize=address /Zi)
 target_link_options(DisplayProtocolTests PRIVATE /INCREMENTAL:NO /DEBUG)
endif()
enable_testing()
add_test(NAME ProtocolValidation COMMAND DisplayProtocolTests)
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque signed TitanVirtualDisplay INF/CAT/DLL package, optional")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 foreach(name TitanVirtualDisplay.inf TitanVirtualDisplay.cat TitanVirtualDisplay.dll)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Incomplete driver package: ${name}")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanVirtualDisplay.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanVirtualDisplay.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanVirtualDisplay.dll\"\n")
endif()
add_library(DisplaySetupCore Installer/Setup.cpp)
target_link_libraries(DisplaySetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32)
add_executable(DisplayPackageTests Tests/PackageTests.cpp)
target_include_directories(DisplayPackageTests PRIVATE Installer)
target_link_libraries(DisplayPackageTests PRIVATE DisplaySetupCore)
add_test(NAME PackageIdentityAndTrust COMMAND DisplayPackageTests)
add_executable(VirtualDisplaySetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
add_executable(VirtualDisplaySetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
foreach(target VirtualDisplaySetup VirtualDisplaySetupConsole)
 target_link_libraries(${target} PRIVATE DisplaySetupCore user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
